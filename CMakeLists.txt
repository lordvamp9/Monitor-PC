cmake_minimum_required(VERSION 3.20)
project(MonitorPC VERSION 2.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Monitor PC requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(hw_monitor WIN32 main.cpp hardware_monitor.cpp resource.rc)
target_compile_definitions(hw_monitor PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(hw_monitor PRIVATE user32 kernel32 gdi32 shell32 advapi32 pdh dxgi)
if(MINGW)
  target_compile_options(hw_monitor PRIVATE -Wall -Wextra)
  target_link_options(hw_monitor PRIVATE -static -static-libgcc -static-libstdc++)
elseif(MSVC)
  target_compile_options(hw_monitor PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(telemetry_tests tests/telemetry_tests.cpp)
  target_include_directories(telemetry_tests PRIVATE .)
  add_test(NAME telemetry COMMAND telemetry_tests)
endif()
